confirm the directory is a repo
$git status
confirm the current branch
$git branch --show-current
check which remote repo is linked
$git remote -v
see what code changed
$git status
to see the actual changes
$git diff
for a shorter summary 
$git diff --stat
check untracked files
$git status --short
check relationship with github
$git fetch origin
$git status
$git branch -vv
stage current changes
$git add .
commit
$git commit -m "Describe the changes"
confirm the commit
$git log -1 --oneline
push to github
$git push
